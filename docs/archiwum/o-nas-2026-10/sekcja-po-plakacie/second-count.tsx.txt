'use client';

import { useRef, type CSSProperties } from 'react';
import type { ONasDict } from '@/lib/i18n/o-nas';
import { Head, Narrator, setNarrator } from './parts';
import { css, lin, seg, usePin } from './shared';
import './second.css';

// SEKCJA PO PLAKACIE - WERSJA 4 „LICZNIK” (2026-10-12; właściciel po wyborze „Plakatu”: „następnie następną sekcję
// dodaj kilka wersji”). Prefiks on-2c-. Przypięta 100svh + 220vh, sam tekst i CSS - bez SVG i WebGL.
// Język „Plakatu” (wielka typografia prowadzona przewijaniem) z własnym motywem: mechaniczny licznik godzin.
// Na cały kadr stoją ogromne cyfry; przy przewijaniu przewijają się jak bębny licznika od 00 do 24 - tyle najdłużej
// czekasz na odpowiedź (jedyna liczba sceny to obronne „24 h”). W ruchu cyfry są samym obrysem („negatyw”, jak
// litery w wejściu), po zatrzymaniu na 24 wypełniają się - bez odbicia. Narrator: jedno zdanie naraz.
// Bez JS / ograniczony ruch: gotowe „24 h” i lista kroków.

const ROW = 0.86; // wysokość wiersza cyfry w em (ta sama liczba stoi w second.css)
const TENS = [0, 1, 2];
const ONES = Array.from({ length: 25 }, (_, i) => i % 10); // 0-9, 0-9, 0-4: jedności jadą bez przerwy do 24

export function SecondCount({ t, live, stepsLabel }: { t: ONasDict['direct']; live: boolean; stepsLabel: string }) {
  const pin = useRef<HTMLDivElement>(null), num = useRef<HTMLDivElement>(null);
  const tens = useRef<HTMLSpanElement>(null), ones = useRef<HTMLSpanElement>(null);
  const narr = useRef<HTMLOListElement>(null);

  usePin(pin, (p) => {
    const x = seg(p, 0.1, 0.78) * 24; // godziny: rozpędza się i hamuje
    // dziesiątki przeskakują jak w liczniku: jadą tylko wtedy, gdy jedności przechodzą z 9 na 0
    const dec = Math.min(2, Math.floor(x / 10)), c = Math.min(1, Math.max(0, x - (dec * 10 + 9)));
    const tp = Math.min(2, dec + c * c * (3 - 2 * c));
    css(ones.current, 'transform', `translateY(${(-x * ROW).toFixed(4)}em)`);
    css(tens.current, 'transform', `translateY(${(-tp * ROW).toFixed(4)}em)`);
    css(num.current, '--f', seg(p, 0.78, 0.9).toFixed(3)); // wypełnienie cyfr po zatrzymaniu (0 = sam obrys)
    setNarrator(narr.current, p < 0.1 ? 0 : p < 0.78 ? 1 : 2, [lin(p, 0, 0.1), lin(p, 0.1, 0.78), lin(p, 0.78, 0.93)]);
  }, live);

  return (
    <section className="on-sec on-2 on-2c" aria-labelledby="on-2-h">
      <Head id="on-2-h" label={t.label} title={t.heading} accent={t.headingAccent} lead={t.leadShort} live={live} />
      <div ref={pin} className="on-pin on-2-pin on-2c-pin">
        <div className="on-stick on-2-stick">
          <div className="container mx-auto px-6 on-2-box on-2c-box">
            <div className="on-2c-stage">
              {/* cyfry są grafiką (bębny licznika); treść dla czytników niżej */}
              <div ref={num} className="on-2c-num" aria-hidden="true">
                <span className="on-2c-col">
                  <span ref={tens} className="on-2c-strip" style={{ '--end': 2 } as CSSProperties}>
                    {TENS.map((d) => <span key={d}>{d}</span>)}
                  </span>
                </span>
                <span className="on-2c-col">
                  <span ref={ones} className="on-2c-strip" style={{ '--end': 24 } as CSSProperties}>
                    {ONES.map((d, i) => <span key={i}>{d}</span>)}
                  </span>
                </span>
                <span className="on-2c-unit">{t.day.unit}</span>
              </div>
              <p className="sr-only">{t.reply}</p>
            </div>
            <Narrator ref={narr} items={t.day.steps} label={stepsLabel} />
          </div>
        </div>
      </div>
    </section>
  );
}
