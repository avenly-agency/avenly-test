// „DLA KOGO PRACUJEMY” - WSPÓLNA WARSTWA „WOW” wersji 5-7 (2026-10-13). Właściciel po czterech pierwszych wersjach:
// „niby spoko, ale potrzebuję czegoś pasującego i oddającego taki efekt wow jak wszędzie”. Pierwszym wersjom brakowało
// tego, co niesie resztę serwisu: żywej mgławicy w barwach klienta, światła i szkła. Tu:
//   - SecSky: mgławica w przypiętej scenie; blask stoi wokół kadru, a BARWA przechodzi płynnie w kolor klienta, który
//     jest na ekranie (czerwień RKS -> złoto Kardysia -> pomarańcz Mcentrum -> błękit marki przy „Twojej firmie”) -
//     jak w sekcji Realizacje na stronie głównej i na /realizacje. Scena sama podaje cel (aim) - bez mierzenia DOM
//     w pętli. Mgławica odsłania się z czerni, gdy scena pierwszy raz wejdzie na ekran,
//   - WowStage: przypięty ekran 100lvh z własną czernią (kompozycja w 100svh - jak wejście), wygaszenia góry i dołu
//     kadru na początku i końcu sceny (mgławica nie jest ucięta prostą krawędzią, gdy scena wjeżdża i odjeżdża),
//   - barwy i akcenty klientów (te same liczby co lib/i18n/home/realizacje.ts i lib/i18n/projects.ts).
// Bez interakcji z kursorem (zasada 2026-10-05). Save-Data / ≤ 2 GB RAM / brak WebGL: sama czerń sceny z poświatą CSS.

export type RGB = [number, number, number];
export type Tint = [RGB, RGB];
/** Barwy mgławicy [jasna, głęboka] klientów - jak `tint` w sekcji Realizacje na stronie głównej. */
const TINTS: Record<string, Tint> = {
  'klub-sportowy': [[0.86, 0.16, 0.13], [0.34, 0.03, 0.08]],
  'grawerstwo-kardys': [[0.86, 0.62, 0.30], [0.42, 0.05, 0.08]],
  mcentrumfizjoterapia: [[1.0, 0.52, 0.08], [0.40, 0.16, 0.02]],
};
const BRAND_TINT: Tint = [[0.23, 0.51, 0.96], [0.08, 0.12, 0.42]];
/** Akcent klienta jako „r g b” (drobne detale: obwódka kadru, strzałka linku) - jak `accent` w lib/i18n/projects.ts. */
const ACCENTS: Record<string, string> = {
  'klub-sportowy': '226 58 50',
  'grawerstwo-kardys': '219 164 92',
  mcentrumfizjoterapia: '251 146 60',
};
/** Barwa mgławicy pozycji (bez nazwy = „Twoja firma” - kolor marki). */
export const tintOf = (slug?: string): Tint => (slug && TINTS[slug]) || BRAND_TINT;
export const accentOf = (slug?: string) => (slug && ACCENTS[slug]) || '59 130 246';

/** Cel mgławicy w ułamkach przypiętego ekranu: środek i połowy wymiarów kadru, wokół którego stoi blask, i barwa. */
export type Aim = { cx: number; cy: number; hw: number; hh: number; tint: Tint };
export type SkyHandle = { aim: (a: Aim) => void };

const REVEAL_MS = 3000, REVEAL_STEP = 34;
const FRAME_FAST = 1000 / 60, FRAME_MS = 1000 / 30;
const SCROLL_HOLD = 240, GLOW_HOLD = 160, GLOW_EPS = 0.003;
const PROBE_OK_MS = 21, PROBE_MIN = 40;
const NO_TRAIL = new Float32Array(32), NO_WAVES = new Float32Array(24);

/** Mgławica sceny. Wzór sterowania: app/(pl)/uslugi/_usluga/sky.tsx (blask liczony w każdej klatce - liveGlow, gaz
    z klatek kluczowych, zegar odsłony krokami, tempo do 60 kl./s tylko gdy sprzęt je utrzymał w czasie odsłony),
    plus płynne przejście barw jak w realizacje/_rl/sky.tsx. */
export function SecSky({ ref }: { ref?: Ref<SkyHandle> }) {
  const hostRef = useRef<HTMLDivElement>(null);
  const goal = useRef<Aim>({ cx: 0.5, cy: 0.5, hw: 0.3, hh: 0.26, tint: BRAND_TINT });
  const wake = useRef<() => void>(() => {});
  useImperativeHandle(ref, () => ({ aim: (a: Aim) => { goal.current = a; wake.current(); } }), []);

  useEffect(() => {
    const host = hostRef.current;
    if (!host) return;
    const nav = navigator as Navigator & { connection?: { saveData?: boolean }; deviceMemory?: number };
    if (nav.connection?.saveData || (nav.deviceMemory && nav.deviceMemory <= 2)) { host.setAttribute('data-fail', ''); return; }
    const touch = window.matchMedia('(pointer: coarse)').matches;
    let neb: NebulaPlus | null = null, raf = 0, dead = false, last = 0, t0 = -1, prev = -1;
    let onScreen = false, scrollAt = -1e9, moveAt = -1e9, rvT = 0;
    let fastOk = false, lastRaf = -1, probing = true;
    const probe: number[] = [];
    const g0 = goal.current;
    const cur = { cx: g0.cx, cy: g0.cy, hw: g0.hw, hh: g0.hh, c1: [...g0.tint[0]] as RGB, c2: [...g0.tint[1]] as RGB };

    function step(now: number) {
      if (!neb) return;
      const dt = prev < 0 ? 1 / 30 : Math.min(0.1, Math.max(0.004, (now - prev) / 1000));
      prev = now;
      const g = goal.current;
      // przed pierwszą narysowaną klatką cel od razu na miejscu - odsłona rozkwita wokół kadru
      const snap = t0 < 0;
      const kf = snap ? 1 : 1 - Math.pow(0.9, dt * 30), kc = snap ? 1 : 1 - Math.pow(0.94, dt * 30);
      let far = Math.abs(g.cx - cur.cx) + Math.abs(g.cy - cur.cy) + Math.abs(g.hw - cur.hw) + Math.abs(g.hh - cur.hh);
      cur.cx += (g.cx - cur.cx) * kf; cur.cy += (g.cy - cur.cy) * kf;
      cur.hw += (g.hw - cur.hw) * kf; cur.hh += (g.hh - cur.hh) * kf;
      for (let i = 0; i < 3; i++) {
        far += Math.abs(g.tint[0][i] - cur.c1[i]) + Math.abs(g.tint[1][i] - cur.c2[i]);
        cur.c1[i] += (g.tint[0][i] - cur.c1[i]) * kc;
        cur.c2[i] += (g.tint[1][i] - cur.c2[i]) * kc;
      }
      // blask albo barwa jeszcze w drodze - pętla trzyma szybkie tempo, żeby przejście było płynne
      if (far > GLOW_EPS) moveAt = now;
      neb.frame(cur.cx, cur.cy, cur.hw, cur.hh);
      neb.interact({
        cx: 0.5, cy: 0.5, lens: 0, trail: NO_TRAIL, waves: NO_WAVES,
        scrollY: window.scrollY, viewH: window.innerHeight, rc: [cur.cx, cur.cy],
      });
      if (t0 >= 0 && rvT < REVEAL_MS) rvT = Math.min(REVEAL_MS, rvT + Math.min(dt * 1000, REVEAL_STEP));
      const rv = t0 < 0 ? 0 : rvT / REVEAL_MS;
      const ok = neb.render(t0 < 0 ? 0 : (now - t0) / 1000, 0.5, 0.5, cur.c1, cur.c2, 1 - Math.pow(1 - rv, 3));
      if (ok && t0 < 0) t0 = now;
    }
    const loop = (now: number) => {
      raf = 0;
      if (dead || document.hidden || !onScreen || !neb) { lastRaf = -1; return; }
      const revealing = t0 < 0 || rvT < REVEAL_MS;
      // próba sprzętu w czasie odsłony: mediana odstępu klatek decyduje, czy potem wolno rysować do 60 kl./s
      if (probing && t0 >= 0) {
        if (revealing) { if (lastRaf > 0 && now - t0 > 500) probe.push(now - lastRaf); }
        else { probing = false; if (probe.length >= PROBE_MIN) { probe.sort((a, b) => a - b); fastOk = probe[probe.length >> 1] <= PROBE_OK_MS; } }
      }
      lastRaf = now;
      const active = now - scrollAt < SCROLL_HOLD || now - moveAt < GLOW_HOLD;
      const gap = revealing || (active && fastOk) ? FRAME_FAST : FRAME_MS;
      if (now - last >= gap - 1) { last = now; step(now); }
      raf = requestAnimationFrame(loop);
    };
    const kick = () => { if (!raf && !dead && neb) raf = requestAnimationFrame(loop); };
    wake.current = kick;
    const resize = () => { if (neb) { neb.resize(host.clientWidth, host.clientHeight); kick(); } };
    const onScroll = () => { scrollAt = performance.now(); };
    neb = createNebulaPlus(host, () => host.setAttribute('data-ready', ''), () => host.setAttribute('data-fail', ''), {
      maxDpr: touch ? (nav.deviceMemory && nav.deviceMemory <= 3 ? 1.25 : 2) : 1.5,
      maxPx: touch ? 1.1e6 : undefined,
      gasScale: touch ? 0.55 : 0.6,
      still: true, gasEvery: 3, gasBusy: 2, liveGlow: true, gasMs: 140,
    });
    resize();
    const ro = new ResizeObserver(resize);
    ro.observe(host);
    const io = new IntersectionObserver((es) => { onScreen = es[es.length - 1].isIntersecting; if (onScreen) kick(); });
    io.observe(host);
    const onVis = () => kick();
    document.addEventListener('visibilitychange', onVis);
    window.addEventListener('scroll', onScroll, { passive: true });
    return () => {
      dead = true;
      wake.current = () => {};
      cancelAnimationFrame(raf);
      ro.disconnect();
      io.disconnect();
      document.removeEventListener('visibilitychange', onVis);
      window.removeEventListener('scroll', onScroll);
      neb?.destroy();
      host.removeAttribute('data-ready');
      host.removeAttribute('data-fail');
    };
  }, []);

  return (
    <div className="on-4-sky" aria-hidden="true">
      <div ref={hostRef} className="on-4-neb" />
    </div>
  );
}

type DivRef = RefObject<HTMLDivElement | null>;

/** Przypięta scena z mgławicą: ekran 100lvh (tło do dołu okna także po schowaniu paska adresu na telefonie),
    kompozycja `fit` 100svh. `len` = długość przewijania w vh. Mgławica tylko przy pełnym ruchu. */
export function WowStage({ pin, stage, fit, sky, len, live, children }: {
  pin: DivRef; stage: DivRef; fit: DivRef; sky: Ref<SkyHandle>; len: number; live: boolean; children: ReactNode;
}) {
  return (
    <div ref={pin} className="on-4-pin" style={{ '--len': len } as CSSProperties}>
      <div ref={stage} className="on-4-stick">
        {live && <SecSky ref={sky} />}
        <div className="on-4-fade on-4-fade--top" aria-hidden="true" />
        <div className="on-4-fade on-4-fade--bot" aria-hidden="true" />
        <div ref={fit} className="on-4-fit">{children}</div>
      </div>
    </div>
  );
}

/** Wygaszenia krawędzi sceny: u góry na początku (scena wjeżdża z czerni strony), u dołu na końcu (odjeżdża w czerń). */
export const edgeFades = (stage: HTMLElement | null, p: number) => {
  if (!stage) return;
  css(stage.querySelector('.on-4-fade--top'), 'opacity', (1 - seg(p, 0, 0.06)).toFixed(3));
  css(stage.querySelector('.on-4-fade--bot'), 'opacity', seg(p, 0.93, 1).toFixed(3));
};

/** Prostokąt elementu w ułamkach przypiętego ekranu (do celu mgławicy). Odczyt układu - wołać tylko przy zmianie
    rozmiaru, nie w klatce. */
export const fractionRect = (el: HTMLElement, stage: HTMLElement) => {
  const r = el.getBoundingClientRect(), s = stage.getBoundingClientRect();
  const w = Math.max(1, s.width), h = Math.max(1, s.height);
  return { cx: (r.left + r.width / 2 - s.left) / w, cy: (r.top + r.height / 2 - s.top) / h, hw: r.width / 2 / w, hh: r.height / 2 / h };
};
