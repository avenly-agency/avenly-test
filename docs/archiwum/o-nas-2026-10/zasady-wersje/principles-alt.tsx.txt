// „W CO WIERZYMY” - PROPOZYCJE 2026-10-13 (właściciel: „teraz wszystkie następne sekcje daj kilka propozycji dla
// każdej”). Obecna wersja („Stacje na linii”, principles.tsx) zostaje domyślna; przełącznik w panelu na dole (dev).
//   PrinciplesStars „Gwiazdozbiór” - przypięta scena (100svh + 5 · 58vh): każda zasada to gwiazda, gwiazda z logo leci
//                   od jednej do drugiej i zostawia linię, aż powstanie cały gwiazdozbiór. Tekst zasady stoi w jednym

// NAGŁÓWEK SEKCJI (runda 29, właściciel: „znowu te headingi wyglądają mediocre”) nie stoi już nad sceną: w „Gwiazdozbiorze”

type Pr = ONasDict['principles'];
const pad = (n: number) => String(n).padStart(2, '0');

/* ───────── GWIAZDOZBIÓR ───────── */

const VB_W = 600, VB_H = 500;
/** Gwiazdy pięciu zasad: wznosząca się łamana (jednostki rysunku 600 × 500). */
const FIVE: [number, number][] = [[64, 430], [196, 318], [322, 392], [438, 206], [540, 84]];
/** Inna liczba zasad: punkty równo rozłożone po tej samej przekątnej, na przemian wyżej i niżej. */
const spread = (n: number): [number, number][] => Array.from({ length: n }, (_, i) => {
  const u = n > 1 ? i / (n - 1) : 0, edge = i === 0 || i === n - 1;
  return [64 + u * 476, 430 - u * 346 + (edge ? 0 : i % 2 ? -46 : 46)];
});
/** Pył gwiazd w tle rysunku - stałe położenia (ten sam wynik na serwerze i w przeglądarce). */
let seed = 11;
const rnd = () => { seed = (seed * 16807) % 2147483647; return seed / 2147483647; };
const DUST = Array.from({ length: 38 }, () => ({
  x: Math.round(rnd() * VB_W), y: Math.round(rnd() * VB_H),
  r: Math.round((0.7 + rnd() * 1.1) * 10) / 10, o: Math.round((0.16 + rnd() * 0.42) * 100) / 100,
}));

export function PrinciplesStars({ t, live }: { t: Pr; live: boolean }) {
  const pin = useRef<HTMLDivElement>(null);
  const glow = useRef<HTMLSpanElement>(null);
  const trav = useRef<SVGGElement>(null);
  const N = t.items.length;
  const pts = N === FIVE.length ? FIVE : spread(N);

  usePin(pin, (p, el) => {
    const pos = p * (N + 1 - 0.35) - 1; // poniżej zera stoi nagłówek sekcji, potem zasady 0..N-1
    // lecąca gwiazda: odcinek k rysuje się na końcu postoju zasady k, gwiazda jest na jego czole
    let x = pts[0][0], y = pts[0][1];
    el.querySelectorAll<SVGPathElement>('.on-p2-link').forEach((ln, k) => {
      const e = seg(pos, k + 0.7, k + 1);
      css(ln, 'stroke-dashoffset', (1 - e).toFixed(4));
      x += (pts[k + 1][0] - pts[k][0]) * e;
      y += (pts[k + 1][1] - pts[k][1]) * e;
    });
    // tekst: jedna zasada naraz - poprzednia odchodzi w górę, zanim gwiazda doleci, następna wchodzi z dołu
    let active = 0, best = -1;
    el.querySelectorAll<HTMLElement>('.on-p2-item').forEach((it, j) => {
      const i = j - 1; // -1 = nagłówek
      const tin = j === 0 ? 1 : seg(pos, i - 0.16, i + 0.04);
      const out = i === N - 1 ? 0 : seg(pos, i + 0.68, i + 0.84);
      const o = tin * (1 - out);
      css(it, 'opacity', o.toFixed(3));
      css(it, 'visibility', o < 0.002 ? 'hidden' : 'visible');
      css(it, 'transform', o >= 0.9995 ? 'none' : `translate3d(0, ${((1 - tin) * 22 - out * 22).toFixed(2)}px, 0)`);
      if (i >= 0 && o > best) { best = o; active = i; }
    });
    el.querySelectorAll<SVGGElement>('.on-p2-node').forEach((nd, i) => {
      const lit = i === 0 ? 1 : seg(pos, i - 0.1, i + 0.02);
      css(nd, 'opacity', (0.3 + 0.7 * lit).toFixed(3));
      flag(nd, 'data-on', i === active && best > 0.5);
    });
    attr(trav.current, 'transform', `translate(${x.toFixed(2)} ${y.toFixed(2)})`);
    css(glow.current, 'left', `${((x / VB_W) * 100).toFixed(2)}%`);
    css(glow.current, 'top', `${((y / VB_H) * 100).toFixed(2)}%`);
  }, live);

  return (
    <section className="on-sec on-p2" aria-labelledby="on-pr-h">
      <div ref={pin} className="on-pin on-p2-pin" style={{ '--n': N + 1 } as CSSProperties}>
        <div className="on-stick on-p2-stick">
          <div className="container mx-auto px-6 on-p2-box">
            <div className="on-p2-texts">
              <Intro
                id="on-pr-h" label={t.label} title={t.heading} accent={t.headingAccent} lead={t.lead} live={live}
                className="on-p2-item on-p2-intro"
              />
              <ol className="on-p2-list">
                {t.items.map((it, i) => (
                  <li key={i} className="on-p2-item">
                    <span className="on-p2-n" aria-hidden="true">{pad(i + 1)} / {pad(N)}</span>
                    <h3 className="on-p2-t">{it.title}</h3>
                    <p className="on-p2-d">{it.desc}</p>
                  </li>
                ))}
              </ol>
            </div>
            <div className="on-p2-sky" aria-hidden="true">
              <svg viewBox={`0 0 ${VB_W} ${VB_H}`} focusable="false">
                {DUST.map((s, i) => <circle key={i} className="on-p2-dust" cx={s.x} cy={s.y} r={s.r} fillOpacity={s.o} />)}
                {pts.slice(0, -1).map((a, k) => (
                  <path key={k} className="on-p2-link" pathLength={1} d={`M${a[0]} ${a[1]}L${pts[k + 1][0]} ${pts[k + 1][1]}`} />
                ))}
                {pts.map((a, i) => (
                  <g key={i} className="on-p2-node" transform={`translate(${a[0]} ${a[1]})`}>
                    <circle className="on-p2-ring" r={30} />
                    <path className="on-p2-gem" d={GEM} transform="scale(.5)" />
                  </g>
                ))}
                <g ref={trav} className="on-p2-trav" transform={`translate(${pts[0][0]} ${pts[0][1]})`}>
                  <path d={GEM} transform="scale(.82)" />
                </g>
              </svg>
              <span
                ref={glow} className="on-p2-glow" data-sky=""
                style={{ left: `${(pts[0][0] / VB_W) * 100}%`, top: `${(pts[0][1] / VB_H) * 100}%` }}
              />
            </div>
          </div>
        </div>
      </div>
    </section>
  );
}

function Card({ i, n, title, desc, live }: { i: number; n: number; title: string; desc: string; live: boolean }) {
  const ref = useRef<HTMLElement>(null);
  useEnter(ref, live, 0.3);
  return (
    <article ref={ref} className="on-p3-card">
      <div className="on-p3-art" aria-hidden="true">{ARTS[i % ARTS.length]}</div>
      <p className="on-p3-n" aria-hidden="true">{pad(i + 1)} / {pad(n)}</p>
      <h3 className="on-p3-t">{title}</h3>
      <p className="on-p3-d">{desc}</p>
    </article>
  );
}

export function PrinciplesCards({ t, live }: { t: Pr; live: boolean }) {

            className="on-p3-head"
          />
          {t.items.map((it, i) => <Card key={i} i={i} n={t.items.length} title={it.title} desc={it.desc} live={live} />)}
        </div>
      </div>
    </section>
  );
}
