  title: string;
  desc: string;
}

export interface UiUxDict {
  breadcrumbBack: string;
  hero: {
    badge: string;

    /** Akcentowana część H1 (gradient span). */
    h1Accent: string;
    lead: string;
    cta: string;
  };
  bento: {
    heading: string;
    lead: string;
    /** 4 karty bento w stałej kolejności (ikony/layout w komponencie). */
    cards: [CardText, CardText, CardText, CardText];
  };
  scope: {
    badge: string;
    /** Część nagłówka przed <br> (z końcową spacją). */
    headingLine1: string;
    /** Część nagłówka po <br> (z wiodącą spacją). */
    headingLine2: string;
    items: CardText[];
  };
}
