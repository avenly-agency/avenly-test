/** Długa jednowyrazowa nazwa: miejsce łamania na wąskim ekranie (jak na planszach listy). */
const BREAK: Record<string, number> = { mcentrumfizjoterapia: 8 };
export const nameOf = (w: WorkItem) => {
  const at = BREAK[w.slug];
  const words = w.name.split(' ');
  const cut = at && words.length === 1 && w.name.length > at ? [w.name.slice(0, at), w.name.slice(at)] : null;
  return {
    node: (cut ? <>{cut[0]}<wbr />{cut[1]}</> : w.name) as ReactNode,
    /** cała nazwa w jednej linijce (od 1024 px) */
    len: w.name.length,
    /** najdłuższy kawałek, gdy nazwa się łamie */
    lenS: Math.max(...(cut ?? words).map((s) => s.length)),
  };
};

/** Zdania tekstu: granica = kropka, spacja i wielka litera („1. miejsce” zostaje w zdaniu). Bez lookbehind - Safari 15. */
export const sentences = (s: string) => {
  const parts = s.split(/([.!?])\s+(?=[A-ZĄĆĘŁŃÓŚŹŻ])/);
  const out: string[] = [];
  for (let i = 0; i < parts.length; i += 2) out.push(parts[i] + (parts[i + 1] ?? ''));
  return out.filter(Boolean);
};
/** Twarda spacja po jednoliterowym słowie („i”, „w”, „z”, „a”…) - pojedyncza litera nie zostaje na końcu linii (telefon). */
export const nb = (s: string) => {
  const one = /(^|[\s(„])([aiouwzAIOUWZ])\s/g;
  return s.replace(one, '$1$2 ').replace(one, '$1$2 '); // drugi przebieg: dwa takie słowa pod rząd („i w”)
};
/** Pierwsze zdanie jako duże zdanie otwierające, reszta jako tekst. */
export const splitLead = (s: string): [string, string] => {
  const [big, ...rest] = sentences(s);
  return [big ?? s, rest.join(' ')];
};

/** Postęp przypiętej sceny 0-1 (ułamkowa pozycja Lenisa - bez schodków na końcu przewinięcia). */
export const pinAt = (sec: HTMLElement, stage: HTMLElement) => {
  const r = sec.getBoundingClientRect();
  return {
    p: clamp01((scrollFrac() - r.top) / Math.max(1, r.height - stage.offsetHeight)),
    near: r.bottom > -200 && r.top < window.innerHeight + 200,
  };
};
export const show = (el: HTMLElement | null, on: boolean) => {
  if (!el) return;
  const v = on ? '' : 'hidden';
  if (el.style.visibility !== v) el.style.visibility = v;
};
/** Własna warstwa tylko na czas ruchu - stała zamraża skalę rastra (po dojeździe kadr i nazwa byłyby miękkie). */
export const moving = (el: HTMLElement | null, on: boolean) => {
  if (!el) return;
  const v = on ? 'transform' : '';
  if (el.style.willChange !== v) el.style.willChange = v;
};
export const scaleX = (el: HTMLElement | null, p: number) => {
  if (!el) return;
  const v = `scaleX(${p.toFixed(4)})`;
  if (el.style.transform !== v) el.style.transform = v;
};
/** Refleks szkła kadru jedzie z przewijaniem - zmienna na samej ramie (na scenie przeliczałaby style całej sceny). */
export const glare = (rim: HTMLElement | null, q: number) => {
  if (!rim) return;
  const v = q.toFixed(3);
  if (rim.style.getPropertyValue('--q') !== v) rim.style.setProperty('--q', v);
};
export const setVar = (el: HTMLElement, name: string, v: string) => { if (el.style.getPropertyValue(name) !== v) el.style.setProperty(name, v); };

// ── PLANSZA / WLOT: plansza tytułowa → kadr → cała strona klienta ──────────────────────────────────
const Film = ({ project, item, num, t, open }: MainProps & { open: 'plansza' | 'wlot' }) => {
  const reduced = useReducedPref();
  const secRef = useRef<HTMLElement>(null);
  const stageRef = useRef<HTMLDivElement>(null);
  const inRef = useRef<HTMLDivElement>(null);
  const plateRef = useRef<HTMLDivElement>(null);
  const titleRef = useRef<HTMLDivElement>(null);
  const metaRef = useRef<HTMLParagraphElement>(null);
  const introRef = useRef<HTMLDivElement>(null);
  const grpRef = useRef<HTMLDivElement>(null);
  const miniRef = useRef<HTMLParagraphElement>(null);
  const frRef = useRef<HTMLDivElement>(null);
  const railRef = useRef<HTMLDivElement>(null);
  const barRef = useRef<HTMLElement>(null);
  const name = nameOf(item);
  const geo = useRef({
    fly: false, tilt: false, css: false, y1: 0, s1: 0.3, down: 500, x0: 0, y0: 0, q0: 0.25, q1: 0.95,
    frame: null as HTMLElement | null, rim: null as HTMLElement | null,
  });

  const frame = useCallback(() => {
    const sec = secRef.current, stage = stageRef.current;
    if (!sec || !stage) return;
    const { p: q, near } = pinAt(sec, stage);
    if (!near) return;
    const g = geo.current;
    // a: 0 = plansza tytułowa, 1 = kadr stoi na swoim miejscu (tuż przed startem strony w kadrze)
    const a = smooth(0.015, g.q0 * 0.96, q);
    const busy = a > 0.001 && a < 0.999;
    const fr = frRef.current;
    moving(fr, busy);
    if (g.fly) {
      // kamera wlatuje w kadr: tekst (bliższy plan) zjeżdża w lewo i gaśnie, kadr prostuje się i rośnie
      const out = smooth(0, 0.6, a);
      put(plateRef.current, `translate3d(${(-a * 160).toFixed(1)}px, 0, 0)`, 1 - out);
      show(plateRef.current, out < 0.995);
      put(miniRef.current, 'none', smooth(0.72, 1, a));
      put(fr, `translate3d(${((1 - a) * g.x0).toFixed(1)}px, ${((1 - a) * g.y0).toFixed(1)}px, 0) rotateY(${((1 - a) * -22).toFixed(2)}deg) rotateX(${((1 - a) * 4).toFixed(2)}deg) scale(${(FLY_SCALE + a * (1 - FLY_SCALE)).toFixed(4)})`);
    } else {
      // kamera odjeżdża: nazwa maleje w lewy górny róg, opis gaśnie, kadr wjeżdża zza dolnej krawędzi
      moving(titleRef.current, busy);
      put(titleRef.current, `translate3d(0, ${(a * g.y1).toFixed(1)}px, 0) scale(${(1 - a * (1 - g.s1)).toFixed(4)})`);
      const meta = metaRef.current;
      if (meta) { const o = clamp01(1 - a * 1.8).toFixed(3); if (meta.style.opacity !== o) meta.style.opacity = o; }
      const out = smooth(0, 0.42, a);
      put(introRef.current, `translate3d(0, ${(-out * 48).toFixed(1)}px, 0)`, 1 - out);
      show(introRef.current, out < 0.995);
      // „Wlot” na wąskim ekranie: kadr leży pochylony w głąb i prostuje się, gdy wjeżdża
      put(fr, `translate3d(0, ${((1 - a) * g.down).toFixed(1)}px, 0)${g.tilt ? ` rotateX(${((1 - a) * 18).toFixed(2)}deg)` : ''} scale(${(0.9 + a * 0.1).toFixed(4)})`);
    }
    put(railRef.current, 'none', smooth(g.q0 * 0.78, g.q0, q));
    // strona klienta w kadrze: przy sda() przesuwa ją CSS na osi przewijania sceny (case.css)
    const pp = clamp01((q - g.q0) / Math.max(0.0001, g.q1 - g.q0));
    if (!g.css) setPage(g.frame, pp);
    scaleX(barRef.current, pp);
    glare(g.rim, q);
  }, []);

  // Geometria sceny: skąd dokąd jedzie nazwa i kadr, długość sceny (wejście + droga strony klienta / SPEED + oddech).
  useEffect(() => {
    const sec = secRef.current, stage = stageRef.current, box = inRef.current, plate = plateRef.current, title = titleRef.current,
      grp = grpRef.current, fr = frRef.current;
    if (reduced || !sec || !stage || !box || !plate || !title || !grp || !fr) return;
    const wide = window.matchMedia(wideMQ), mid = window.matchMedia('(min-width: 640px)');
    const page = fr.querySelector<HTMLElement>('.rl-page');
    const measure = () => {
      const fly = open === 'wlot' && wide.matches;
      const nameEl = title.querySelector<HTMLElement>('.rl-cf-name');
      const fs = nameEl ? parseFloat(getComputedStyle(nameEl).fontSize) : 120;
      const s1 = Math.min(1, (wide.matches ? 38 : mid.matches ? 32 : 26) / fs);
      // miejsce na małą nazwę nad kadrem
      setVar(box, '--tf', `${fly ? miniRef.current?.offsetHeight || 46 : Math.round(title.offsetHeight * s1)}px`);
      const view = fr.querySelector<HTMLElement>('.rl-frame-view');
      const travel = view && page ? Math.max(0, page.offsetHeight - view.clientHeight) : 0;
      const vh = stage.offsetHeight;
      const intro = vh * 0.9, run = travel / SPEED;
      const d = Math.round(intro + run + vh * 0.16);
      setVar(sec, '--cf-d', `${d}px`);
      setVar(sec, '--cf-p0', `${((intro / d) * 100).toFixed(3)}%`);
      setVar(sec, '--cf-p1', `${(((intro + run) / d) * 100).toFixed(3)}%`);

export const Stack = ({ project, item, t }: StoryProps) => {
  const ref = useRef<HTMLDivElement>(null);
  const reduced = useReducedPref();
  const parts = useRef<{ card: HTMLElement; shade: HTMLElement | null; world: HTMLElement | null }[]>([]);
  const step = useRef(58);
  const shots = shotsOf(project);

  // Wysokości kart: z treści, a karta z przodu nigdy krótsza niż wystająca spod niej karta z tyłu (inaczej dół
  // przykrytej karty wystawałby pod przednią). `--h` = wysokość dla przyklejenia (wysoka karta przykleja się dołem).
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const cards = Array.from(el.querySelectorAll<HTMLElement>('.rl-cf-card'));
    const ins = cards.map((c) => c.querySelector<HTMLElement>('.rl-cf-card-in'));
    parts.current = cards.map((card) => ({
      card, shade: card.querySelector<HTMLElement>('.rl-cf-shade'), world: card.querySelector<HTMLElement>('.rl-cf-world-in'),
    }));
    const measure = () => {
      const cs = getComputedStyle(el);
      const st0 = parseFloat(cs.getPropertyValue('--cf-st0')) || 96;
      const stp = parseFloat(cs.getPropertyValue('--cf-step')) || 58;
      step.current = stp;
      const vh = window.innerHeight;
      ins.forEach((n) => { if (n && n.style.minHeight) n.style.minHeight = ''; });
      const nat = cards.map((c) => c.offsetHeight);
      let low = 0;
      nat.forEach((h, i) => {
        const at = st0 + i * stp;
        let hh = h;
        if (Math.min(at, vh - h - 12) + h < low) hh = Math.ceil(low - at);
        low = Math.max(low, Math.min(at, vh - hh - 12) + hh);
        const n = ins[i];
        if (n && hh !== h) n.style.minHeight = `${hh}px`;
        cards[i].style.setProperty('--h', `${hh}px`);
      });
    };
    measure();
    const ro = new ResizeObserver(measure);
    el.querySelectorAll('.rl-cf-card-tx').forEach((tx) => ro.observe(tx));
    window.addEventListener('resize', measure);
    document.fonts?.ready.then(measure);
    el.setAttribute('data-stack', '');
    return () => {
      ro.disconnect();
      window.removeEventListener('resize', measure);
      el.removeAttribute('data-stack');
      ins.forEach((n) => { if (n) n.style.minHeight = ''; });
      parts.current = [];
    };
  }, []);

  useScrollLoop(() => {
    const el = ref.current, list = parts.current;
    if (!el || !list.length) return;
    const vh = window.innerHeight;
    const r = el.getBoundingClientRect();
    if (r.bottom < -200 || r.top > vh + 200) return;
    const f = scrollFrac();
    // odczyty przed zapisami
    const tops = list.map((c) => c.card.getBoundingClientRect().top - f);
    const hs = list.map((c) => c.card.offsetHeight);
    let depth = 0;
    for (let i = list.length - 1; i >= 0; i--) {
      // o ile następna karta nasunęła się na tę (1 = widać już tylko zakładkę)
      if (i < list.length - 1) depth += clamp01((tops[i] + hs[i] - tops[i + 1]) / Math.max(1, hs[i] - step.current));
      const c = list[i];
      put(c.card, depth > 0.0005 ? `scale(${(1 - 0.035 * depth).toFixed(4)})` : 'none');
      if (c.shade) { const o = Math.min(0.64, depth * 0.34).toFixed(3); if (c.shade.style.opacity !== o) c.shade.style.opacity = o; }
      // kadry w głębi karty jadą trochę wolniej niż karta (tylko gdy karta wjeżdża - przyklejona stoi)
      if (c.world) put(c.world, `translate3d(0, ${(((tops[i] + hs[i] / 2) / vh - 0.5) * 44).toFixed(1)}px, 0)`);
    }
  }, !reduced, []);

  return (
    <div ref={ref} className="rl-cf-stack">
      <StackCard i={0} no="01" label={t.challengeLabel} world={<VoidShot item={item} className="rl-cf-shot" />}>
        <Copy text={project.challenge} />
      </StackCard>
      <StackCard
        i={1} no="02" label={t.solutionLabel} flip
        world={shots.length
          ? shots.map((src, k) => <Shot key={src} item={item} src={src} className={`rl-cf-shot${k ? ' rl-cf-shot--back' : ''}`} />).reverse()
          : <Shot item={item} className="rl-cf-shot" />}
      >
        <Copy text={project.solution} />
      </StackCard>
      <StackCard i={2} no="03" label={t.statsLabel}>
        <Facts project={project} item={item} t={t} />
      </StackCard>
    </div>
  );
};

/** Pozycja kamery (numer planszy, ułamek = w drodze). `keys` = końce odcinków w px przewijania, na przemian
    postój i przejazd: [koniec postoju 0, koniec przejazdu 0→1, koniec postoju 1, ...]. */
export const camAt = (x: number, keys: number[]) => {
  let from = 0;
  for (let i = 0; i < keys.length; i++) {
    if (x <= keys[i]) return i % 2 === 0 ? i / 2 : (i - 1) / 2 + smooth(from, keys[i], x);
    from = keys[i];
  }
  return Math.floor(keys.length / 2);
};

      const leave = k === n - 1 ? 0 : smooth(k === 0 ? 0.7 : 0.78, 1, x);
      const o = enter * (1 - leave);
      put(el, `translate3d(0, ${((1 - enter) * 26 - leave * 26).toFixed(1)}px, 0)`, o);
      show(el, o > 0.004);
    });
    // strona klienta gra przez całą opowieść: rusza, gdy nazwa schodzi z ekranu
    const pp = clamp01((u - 0.6) / Math.max(0.5, n - 1.1));
    if (!g.css) setPage(g.frame, pp);
    scaleX(barRef.current, pp);
    glare(g.rim, q);
  }, []);

  useEffect(() => {
    const sec = secRef.current, stage = stageRef.current, nar = narRef.current, fr = frRef.current;
    if (reduced || !sec || !stage || !nar || !fr) return;
    const measure = () => {
      const items = Array.from(nar.children) as HTMLElement[];
      const n = items.length;
      const vh = stage.offsetHeight;
      const d = Math.round(vh * UNIT * n);
      setVar(sec, '--cf-d', `${d}px`);
      setVar(sec, '--cf-p0', `${((0.6 / n) * 100).toFixed(3)}%`);
      setVar(sec, '--cf-p1', `${(((n - 0.5) / n) * 100).toFixed(3)}%`);
      geo.current = {
        css: sda(), per: n, items,
        frame: fr.querySelector<HTMLElement>('.rl-frame'), rim: fr.querySelector<HTMLElement>('.rl-frame-rim'),
      };
      frame();
    };
    measure();
    const ro = new ResizeObserver(measure);
    ro.observe(stage);
    const late = window.setTimeout(measure, 60);
    return () => {
      ro.disconnect();
      window.clearTimeout(late);
      ['--cf-d', '--cf-p0', '--cf-p1'].forEach((v) => sec.style.removeProperty(v));
    };
  }, [reduced, frame]);

  useScrollLoop(frame, !reduced, []);

  return (
    <section ref={secRef} className="rl-cf-se rl-cf-sda" aria-labelledby="rl-cf-h1">
      <div ref={stageRef} className="rl-cf-se-stage">
        <div className="rl-cf-se-box container mx-auto px-6">
          <div ref={frRef} className="rl-cf-se-fr">
            <LiveFrame item={item} t={t} />
            <i className="rl-cf-bar rl-cf-se-bar" aria-hidden="true"><b ref={barRef} /></i>
          </div>
          <div ref={narRef} className="rl-cf-nar">
            <div className="rl-cf-nar-i rl-cf-nar-t" style={{ '--len-s': name.lenS } as CSSProperties}>
              <Meta num={num} item={item} year={project.year} />
              <Name item={item} className="rl-cf-nar-name" />
              <p className="im-lead rl-cf-lead rl-in" style={{ '--d': 4 } as CSSProperties}>{project.description}</p>
            </div>
            {before.map((s, i) => <Said key={`a${i}`} no="01" label={t.challengeLabel} first={i === 0} text={s} />)}
            {after.map((s, i) => <Said key={`b${i}`} no="02" label={t.solutionLabel} first={i === 0} text={s} />)}
          </div>
        </div>
      </div>
    </section>
  );
};
