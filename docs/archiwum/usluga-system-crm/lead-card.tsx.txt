'use client';

import { useCallback, useEffect, useRef, type CSSProperties, type ReactNode } from 'react';
import { Bell, FileText, Inbox, PhoneCall, StickyNote, type LucideIcon } from 'lucide-react';
import type { LeadCopy, LeadEventKind } from '@/lib/i18n/uslugi/system-crm-lead';
import { seg, usePin, useReducedPref } from '../../_usluga/shared';
import { CHANNEL_ICON, LeadPin, Reel, bloom, clearSteps, flag, mark, narrate, pinProgress, put, stepsOf, unbloom, useFit, useKick, useRise, wipe } from './lead-kit';
import type { SceneProps } from './types';
import { Av, Win } from './ui';
import './lead.css';

// SEKCJA „NIC JUŻ NIE GINIE”, wersja „KARTA KLIENTA” - przypięta (300vh), sterowana wyłącznie przewijaniem.
// Bohaterem jest jedna karta klienta - to, co ma każdy CRM: osoba i cała historia kontaktu w jednym miejscu.
// WSZYSTKO, CO SIĘ POJAWIA ALBO PODMIENIA, ROBI TO ROZEJŚCIEM (.cr-bloom / .cr-bloom-out z system-crm.css; decyzja
// właściciela 2026-10-05: „tak jak się nebula rozchodzi” - bez linii światła, kurtyn i przycięć prostą krawędzią):
//   Trzy wiadomości  w historii leży zapytanie z formularza; pod nim, od lewej krawędzi (stamtąd przychodzą),
//                    rozchodzą się e-mail i nieodebrany telefon od tej samej osoby. Po lewej wciąż „Nowy kontakt”.
//   Jedna karta      dwie wiadomości wsuwają się pod pierwszą (prawdziwy ruch - znikają za nią), jej treść
//                    przewija się na „Pierwszy kontakt”, zapalają się znaczki trzech kanałów. Potem od strony
//                    pierwszego wpisu rozchodzi się karta klienta (nazwisko, etap, opiekun, następny krok),
//                    a „Nowy kontakt” znika jej dokładnym dopełnieniem.
//   Historia         na osi (statyczna włosowa linia) dochodzą kolejne wpisy - każdy rozchodzi się od swojego
//                    znaczka na osi: rozmowa, notatka, oferta; etap i następny krok obracają się jak bębny.
//   Przypomnienie    klient milczy - na osi rozchodzi się przypomnienie w kolorze podstrony, następny krok zmienia
//                    się sam.
// Jedno miejsce akcji naraz: strona okna, na której nic się nie dzieje, jest przygaszona (data-focus).
// Geometria w CSS: wpisy mają stałą wysokość (--eh) i odstęp (--eg), wiadomości 2 i 3 leżą na miejscach wpisów
// 2 i 3 i wsuwają się o wielokrotność (--eh + --eg) - scena niczego nie mierzy poza dopasowaniem okna do pola.
// Zmienne: na obiekcie --zin (najazd); na wnętrzu okna --sg (etap), --nx (następny krok); --bloom + data-bloomed na
// wiadomościach, wpisach i obu warstwach karty klienta; na wiadomości --m (wsunięcie), na pierwszym wpisie --mg
// (zmiana treści) i data-got (ile wiadomości już wchłonął).
// Bez JS / ograniczony ruch: zwykła sekcja - gotowa karta klienta z kompletną historią, zdania jako lista.

const LEN = 300;
/** Oś sceny (postęp przypięcia 0-1): najazd, rozejście wiadomości 2 i 3, wsunięcie ich pod pierwszą, zmiana treści
    pierwszego wpisu, rozejście karty klienta, kolejne wpisy historii, zmiany etapu i następnego kroku. */
const ZIN: [number, number] = [0, 0.07];
const LAND: [number, number][] = [[0.08, 0.16], [0.18, 0.26]];
const MERGE: [number, number][] = [[0.33, 0.4], [0.39, 0.46]];
const FLIP: [number, number] = [0.42, 0.48];
const IDENT: [number, number] = [0.48, 0.58];
const EVENTS: [number, number][] = [[0.6, 0.67], [0.67, 0.74], [0.74, 0.81], [0.86, 0.94]];
const STAGE: [number, number][] = [[0.65, 0.69], [0.79, 0.83]];
const NEXT: [number, number][] = [[0.72, 0.76], [0.91, 0.95]];
const CUTS = [0.31, 0.59, 0.845];

const EVENT_ICON: Record<LeadEventKind, LucideIcon> = { call: PhoneCall, note: StickyNote, offer: FileText, remind: Bell };

interface Els { first: HTMLElement | null; slips: HTMLElement[]; events: HTMLElement[]; ident: HTMLElement[]; steps: HTMLElement[] }

const Fact = ({ label, children }: { label: string; children: ReactNode }) => (
  <div className="cr7-a-fact"><dt>{label}</dt><dd>{children}</dd></div>
);

export const LeadCard = ({ t, x, headId }: SceneProps<LeadCopy>) => {
  const c = x.card;
  const person = t.team[c.who] ?? t.team[0];
  const [msg, ...more] = c.messages;
  const reduced = useReducedPref();
  const ref = useRef<HTMLElement>(null);
  const viewRef = useRef<HTMLDivElement>(null);
  const objRef = useRef<HTMLDivElement>(null);
  const bodyRef = useRef<HTMLDivElement>(null);
  const els = useRef<Els>({ first: null, slips: [], events: [], ident: [], steps: [] });
  useRise(viewRef, !reduced);
  useFit(viewRef, objRef, !reduced);

  const draw = useCallback((p: number) => {
    const obj = objRef.current, body = bodyRef.current, e = els.current;
    if (!obj || !body) return;
    const zin = seg(p, ZIN[0], ZIN[1]);
    put(obj, '--zin', zin.toFixed(4));
    flag(obj, 'data-flat', zin >= 1);
    // wiadomości 2 i 3: rozchodzą się od lewej, potem wsuwają się pod pierwszy wpis
    let got = 0;
    e.slips.forEach((li, k) => {
      const [la, lb] = LAND[k] ?? LAND[LAND.length - 1];
      const [ma, mb] = MERGE[k] ?? MERGE[MERGE.length - 1];
      const m = seg(p, ma, mb);
      bloom(li, seg(p, la, lb));
      put(li, '--m', m.toFixed(4));
      flag(li, 'data-gone', m >= 1);
      if (m > 0.85) got += 1;
    });
    if (e.first) {
      put(e.first, '--mg', seg(p, FLIP[0], FLIP[1]).toFixed(4));
      mark(e.first, 'data-got', String(got));
    }
    // kolejne wpisy historii: każdy rozchodzi się od swojego znaczka na osi
    e.events.forEach((li, j) => {
      const [a, b] = EVENTS[j] ?? EVENTS[EVENTS.length - 1];
      bloom(li, seg(p, a, b));
    });
    // karta klienta rozchodzi się, „Nowy kontakt” znika jej dopełnieniem (ten sam postęp na obu warstwach)
    const idv = seg(p, IDENT[0], IDENT[1]);
    e.ident.forEach((el) => bloom(el, idv));
    put(body, '--sg', STAGE.reduce((sum, [a, b]) => sum + seg(p, a, b), 0).toFixed(4));
    put(body, '--nx', NEXT.reduce((sum, [a, b]) => sum + seg(p, a, b), 0).toFixed(4));
    // jedno miejsce akcji: najpierw historia, potem karta klienta, dalej całe okno
    mark(body, 'data-focus', p < IDENT[0] ? 'hist' : p < IDENT[1] + 0.02 ? 'id' : 'all');
    narrate(e.steps, p, CUTS);
  }, []);

  useEffect(() => {
    const root = ref.current, obj = objRef.current, body = bodyRef.current;
    if (!root || !obj || !body) return;
    const e: Els = {
      first: body.querySelector<HTMLElement>('.cr7-a-first'),
      slips: Array.from(body.querySelectorAll<HTMLElement>('.cr7-a-slip')),
      events: Array.from(body.querySelectorAll<HTMLElement>('.cr7-a-ev[data-kind]')),
      ident: Array.from(body.querySelectorAll<HTMLElement>('.cr7-a-idnew, .cr7-a-idcard')),
      steps: stepsOf(root),
    };
    els.current = e;
    if (reduced) {
      // stan końcowy z układu „bez JS” - zdejmujemy wszystko, co scena zdążyła zapisać
      wipe(obj, ['--zin'], ['data-flat']);
      wipe(body, ['--sg', '--nx'], ['data-focus']);
      wipe(e.first, ['--mg'], ['data-got']);
      e.slips.forEach((li) => { unbloom(li); wipe(li, ['--m'], ['data-gone']); });
      [...e.events, ...e.ident].forEach(unbloom);
      clearSteps(e.steps);
    }
  }, [reduced]);

  usePin(ref, draw, !reduced);
  const kick = useCallback(() => { if (ref.current) draw(pinProgress(ref.current)); }, [draw]);
  useKick(kick, !reduced);

  const MsgIcon = CHANNEL_ICON[msg.ch];
  return (
    <LeadPin rootRef={ref} kind="a" len={LEN} headId={headId} x={x} steps={c.steps} stepsAria={t.film.stepsAria}>
      <div ref={viewRef} className="cr7-fitview">
        <div ref={objRef} className="cr7-obj cr7-a-obj">
          <div className="cr7-rise cr-bloom">
            <div className="cr7-a-cam">
              <Win app={t.app} className="cr7-a-win">
                <div ref={bodyRef} className="cr7-a">
                  <div className="cr7-a-id">
                    <div className="cr7-a-idin">
                      {/* zanim system rozpozna osobę: „Nowy kontakt” (warstwa, która schodzi - tylko w scenie) */}
                      <div className="cr7-a-idnew cr-bloom-out" aria-hidden="true">
                        <span className="cr7-a-face">?</span>
                        <span className="cr7-a-who"><b>{c.unknown.name}</b><span>{c.unknown.meta}</span></span>
                        <dl className="cr7-a-facts">
                          <Fact label={c.facts.stage}>{x.stages[0]}</Fact>
                          <Fact label={c.facts.owner}>{c.unknown.none}</Fact>
                          <Fact label={c.facts.next}>{c.unknown.none}</Fact>
                        </dl>
                      </div>
                      <div className="cr7-a-idcard cr-bloom">
                        <span className="cr7-a-face" aria-hidden="true">{c.client.charAt(0)}</span>
                        <span className="cr7-a-who"><b>{c.client}</b><span>{c.meta}</span></span>
                        <dl className="cr7-a-facts">
                          <Fact label={c.facts.stage}><Reel className="cr7-a-stage" items={x.stages.slice(0, 3)} /></Fact>
                          <Fact label={c.facts.owner}><Av name={person.name} />{person.name}</Fact>
                          <Fact label={c.facts.next}><Reel className="cr7-a-next" items={c.next} /></Fact>
                        </dl>
                      </div>
                    </div>
                  </div>
                  <div className="cr7-a-hist">
                    <p className="cr7-a-hist-h">{c.historyTitle}</p>
                    <ol className="cr7-a-list" aria-label={c.historyAria}>
                      {/* pierwszy wpis: zapytanie z formularza, które wchłania dwie kolejne wiadomości tej samej osoby */}
                      <li className="cr7-a-ev cr7-a-first">
                        <span className="cr7-a-flip">
                          <span className="cr7-a-flip-in">
                            <span aria-hidden="true">
                              <span className="cr7-a-ico"><MsgIcon /></span>
                              <span className="cr7-a-txt"><b>{x.channels[msg.ch]}</b><span>{msg.text}</span></span>
                            </span>
                            <span>
                              <span className="cr7-a-ico"><Inbox aria-hidden="true" /></span>
                              <span className="cr7-a-txt"><b>{c.merged.title}</b><span>{c.merged.text}</span></span>
                            </span>
                          </span>
                        </span>
                        <span className="cr7-a-chs" aria-hidden="true">
                          {c.messages.map((m) => {
                            const Icon = CHANNEL_ICON[m.ch];
                            return <i key={m.ch}><Icon /></i>;
                          })}
                        </span>
                      </li>
                      {more.map((m, k) => {
                        const Icon = CHANNEL_ICON[m.ch];
                        return (
                          <li key={m.ch} className="cr7-a-ev cr7-a-slip cr-bloom" style={{ '--k': k + 1 } as CSSProperties} aria-hidden="true">
                            <span className="cr7-a-ico"><Icon /></span>
                            <span className="cr7-a-txt"><b>{x.channels[m.ch]}</b><span>{m.text}</span></span>
                            <span className="cr7-a-time">{m.time}</span>
                          </li>
                        );
                      })}
                      {c.events.map((ev) => {
                        const Icon = EVENT_ICON[ev.kind];
                        return (
                          <li key={ev.kind} className="cr7-a-ev cr-bloom" data-kind={ev.kind}>
                            <span className="cr7-a-ico"><Icon aria-hidden="true" /></span>
                            <span className="cr7-a-txt"><b>{ev.title}</b><span>{ev.text.replace('{name}', person.name)}</span></span>
                            <span className="cr7-a-time">{ev.time}</span>
                          </li>
                        );
                      })}
                    </ol>
                  </div>
                </div>
              </Win>
            </div>
          </div>
        </div>
      </div>
    </LeadPin>
  );
};
