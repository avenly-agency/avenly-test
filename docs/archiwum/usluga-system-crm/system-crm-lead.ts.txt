// miejscu, karta klienta z historią, sprawa w etapach o ogólnych nazwach, system sam przypisuje osobę, termin
// i przypomina o sprawie, która stoi. Żadnych słów w rodzaju szafa, pomiar, montaż. AI tu nie występuje.
// Zasady (PRODUCT.md): głos korzyści, sentence case, bez myślników em / en, bez liczb wyglądających jak wyniki i bez
// cen; klienci to zwykłe imiona z inicjałem albo neutralna nazwa firmy z głównego słownika („Biuro Linia”).
// `{name}` w tekstach = imię osoby z zespołu (zdania ułożone tak, żeby nie zależały od rodzaju).
// Twarde spacje w PL dokłada koordynator (`typo()` w system-crm.ts).

export type LeadChannel = 'form' | 'mail' | 'phone';
export type LeadEventKind = 'call' | 'note' | 'offer' | 'remind';

export interface LeadCopy {
  /** Nagłówek sekcji: zdanie bez kropki (kropkę w kolorze podstrony dokłada komponent). */
  title: string;
  lead: string;
  /** Etapy sprawy - ogólne, takie same w każdej firmie. „Karta klienta” używa trzech pierwszych. */
  stages: string[];
  stagesAria: string;
  /** Kanały, którymi przychodzą zapytania. */
  channels: Record<LeadChannel, string>;
  /** Wersja „Kanały” (lead-channels.tsx): trzy źródła, jedna lista. `who` = indeks w `t.team`, `due` = termin,
      który system ustawia sam, `done` = to, co system zrobił na koniec. Zdań narratora dokładnie trzy. */
  inbox: {
    steps: string[];
    sourcesAria: string;
    listTitle: string;
    listAria: string;
    rows: { ch: LeadChannel; name: string; time: string; text: string; who: number; due: string; done: string }[];
  };
  /** Wersja „Lejek” (lead-funnel.tsx): `cases[i]` = sprawy na etapie i (`age` = od kiedy bez ruchu). Sprawa-bohater
      (`hero`) stoi jako ostatnia na etapie 2 („Oferta”) i rusza na etap 3 - nie ma jej w `cases`. `days` = licznik
      dni bez ruchu (ostatnia pozycja = stan po ruszeniu), `text` / `idle` = opis i stan na postoju, `textDone` /
      `idleDone` = po ruszeniu. Etapy 2 i 3 mają w `cases` tyle samo spraw - wtedy sprawa jedzie prosto w bok. */
  funnel: {
    steps: string[];
    title: string;
    cases: { name: string; age: string }[][];
    hero: { name: string; who: number; days: string[]; text: string; textDone: string; idle: string; idleDone: string };
    note: { label: string; time: string; text: string; done: string };
  };
  /** Wersja „Karta klienta” (lead-card.tsx): `messages` = trzy wiadomości tej samej osoby (pierwsza zostaje jako wpis,
      dwie kolejne się w nią wsuwają), `merged` = wpis po sklejeniu, `events` = kolejne wpisy historii (dokładnie
      cztery, ostatni = przypomnienie), `next` = „następny krok” po kolei (trzy pozycje). */
  card: {
    steps: string[];
    client: string;
    meta: string;
    who: number;
    unknown: { name: string; meta: string; none: string };
    facts: { stage: string; owner: string; next: string };
    next: string[];
    historyTitle: string;
    historyAria: string;
    messages: { ch: LeadChannel; text: string; time: string }[];
    merged: { title: string; text: string };
    events: { kind: LeadEventKind; title: string; text: string; time: string }[];
  };
}
